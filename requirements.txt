flask==3.0.3
requests==2.34.2
# Reclaiming leaked preview servers (workspace.py _kill_listener /
# sweep_own_range). NOT optional in practice: without psutil that code returns
# False and reclaims NOTHING, so every preview that outlives its hub keeps its
# port forever. Measured on this machine 2026-08-30: 18 orphaned
# `python -m http.server` processes, 7 of them still holding ports 5800-5806
# and refusing connections -- which is what a user sees as "all pages 404".
# The import is guarded, so a missing psutil fails silently rather than loudly;
# that is exactly why it has to be pinned here.
psutil==7.2.1

# WebP re-encoding of generated images (app.py _to_webp_b64). Optional at
# runtime -- the hub falls back to the generator's own format if it is
# missing -- but pinned here so every install actually gets WebP.
Pillow==10.2.0

# AES-256-GCM for the provider API keys stored in config.json (secretstore.py).
# NOT optional, despite the guarded import that makes it look that way.
#
# THIS LINE'S ABSENCE COST AN INSTALL EVERY API KEY IT HAD. run.bat creates a
# venv and installs THIS FILE into it; cryptography was not here, so a hub
# started through the launcher had no AESGCM, could not decrypt a single stored
# key, and -- before the ciphertext carry-back existed -- the next save wrote
# the shorter list. All 41 keys at once. The all-or-nothing shape is the
# signature: a missing MODULE fails every key simultaneously, which is why it
# looked nothing like a file race and took a whole session to find.
#
# Same reasoning as psutil above, and more severe: the import is guarded so a
# missing package fails SILENTLY, which is exactly why it has to be pinned
# here. config.py now also says so out loud when it cannot read a key.
cryptography==50.0.1

# Hermes Connect/Disconnect (app._autofix_hermes / _disconnect_hermes /

# worse -- Disconnect silently leaves the hub's model block in Hermes' config.
# MEASURED 2026-09-30: the hub runs from the .venv, which had no yaml; the
# tests only passed under the system Python, which happened to have it.
PyYAML==6.0.2
